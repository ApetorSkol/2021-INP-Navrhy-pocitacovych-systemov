library ieee;
use ieee.std_logic_1164.all;
use ieee.std_logic_arith.all;
use ieee.std_logic_unsigned.all;

-- ----------------------------------------------------------------------------
--                        Entity declaration
-- ----------------------------------------------------------------------------
entity cpu is
 port (
   CLK   : in std_logic;  -- hodinovy signal
   RESET : in std_logic;  -- asynchronni reset procesoru
   EN    : in std_logic;  -- povoleni cinnosti procesoru
 
   -- synchronni pamet ROM
   CODE_ADDR : out std_logic_vector(11 downto 0); -- adresa do pameti
   CODE_DATA : in std_logic_vector(7 downto 0);   -- CODE_DATA <- rom[CODE_ADDR] pokud CODE_EN='1'
   CODE_EN   : out std_logic;                     -- povoleni cinnosti
   
   -- synchronni pamet RAM
   DATA_ADDR  : out std_logic_vector(9 downto 0); -- adresa do pameti
   DATA_WDATA : out std_logic_vector(7 downto 0); -- ram[DATA_ADDR] <- DATA_WDATA pokud DATA_EN='1'
   DATA_RDATA : in std_logic_vector(7 downto 0);  -- DATA_RDATA <- ram[DATA_ADDR] pokud DATA_EN='1'
   DATA_WREN  : out std_logic;                    -- cteni z pameti (DATA_WREN='0') / zapis do pameti (DATA_WREN='1')
   DATA_EN    : out std_logic;                    -- povoleni cinnosti
   
   -- vstupni port
   IN_DATA   : in std_logic_vector(7 downto 0);   -- IN_DATA obsahuje stisknuty znak klavesnice pokud IN_VLD='1' a IN_REQ='1'
   IN_VLD    : in std_logic;                      -- data platna pokud IN_VLD='1'
   IN_REQ    : out std_logic;                     -- pozadavek na vstup dat z klavesnice
   
   -- vystupni port
   OUT_DATA : out  std_logic_vector(7 downto 0);  -- zapisovana data
   OUT_BUSY : in std_logic;                       -- pokud OUT_BUSY='1', LCD je zaneprazdnen, nelze zapisovat,  OUT_WREN musi byt '0'
   OUT_WREN : out std_logic                       -- LCD <- OUT_DATA pokud OUT_WE='1' a OUT_BUSY='0'
 );
end cpu;


-- ----------------------------------------------------------------------------
--                      Architecture declaration
-- ----------------------------------------------------------------------------
architecture behavioral of cpu is
	-- deklaracia pre PROGRAM COUNTER
	signal prog_count_reg : std_logic_vector(11 downto 0);
	signal prog_count_inc : std_logic;
	signal prog_count_dec : std_logic;

	-- deklaracia pre register POINTER
	signal pointer_reg : std_logic_vector(9 downto 0);
	signal pointer_inc : std_logic;
	signal pointer_dec : std_logic;

	--deklaracia pre MULTIPLEXOR
	signal multi_out :  std_logic_vector(7 downto 0);
	signal multi_sel :  std_logic_vector(1 downto 0);

	--deklaracia temp pre cyklus
	signal temp :  std_logic_vector(7 downto 0);

	--mnozina vsetkych stavov automatu
	type fsm_state is (
		--zakladne stavy automatu
		state_start,
		state_fetch,
		state_decode,
		state_halt,
		
		-- instrukcie <>
		state_pointer_inc,
		state_pointer_dec,
		-- instrukcie +-
		
		state_value_inc,
		state_value_inc_2,
		state_value_inc_3,
		state_value_inc_4,
		state_value_dec,
		state_value_dec_2,
		state_value_dec_3,
		state_value_dec_4,

		-- instrukcie getchar a putchar
		state_putchar,
		state_putchar_2,
		state_getchar,
		state_getchar_2,

		-- cykly
		state_cycle_start,
		state_cycle_start_1,
		state_cycle_start_2,
		state_cycle_end,
		state_cycle_end_1,
		state_cycle_end_2,
		state_cycle_end_3,
		state_cycle_end_4,
		state_cycle_break,
		state_cycle_break_0,
		state_cycle_break_1
	);
	signal state : fsm_state := state_start;
	signal next_state : fsm_state ;

begin


	prog_count: process (RESET, CLK, prog_count_inc , prog_count_dec) is
	begin
		if (RESET='1') then
			prog_count_reg <= (others=>'0');
		elsif (CLK'event) and (CLK='1') then
			if prog_count_inc = '1' then
				prog_count_reg <= prog_count_reg + 1;
			elsif prog_count_dec = '1' then
				prog_count_reg <= prog_count_reg - 1;
			end if;
		end if;
		CODE_ADDR <= prog_count_reg ; 
	end process;


	pointer_count: process (RESET, CLK, pointer_inc , pointer_dec) is
	begin
		if (RESET='1') then
			pointer_reg <= (others=>'0');
		elsif (CLK'event) and (CLK='1') then
			if pointer_inc = '1' then
				pointer_reg <= pointer_reg + 1;
			elsif pointer_dec = '1' then
				pointer_reg <= pointer_reg - 1;
			end if;
		end if;
		DATA_ADDR <= pointer_reg;
	end process;

	

	multiplexor: process (RESET, CLK, multi_sel) is

	begin
		if (RESET='1') then
			multi_out <= (others=>'0');
		elsif (CLK'event) and (CLK='1') then
			case multi_sel is

				when "00" =>
					multi_out <= IN_DATA;
				when "01" =>
					multi_out <= DATA_RDATA - 1 ;
				when "10" =>
					multi_out <= DATA_RDATA + 1 ;
				when others =>
					multi_out <= (others=>'0');
			end case;
			DATA_WDATA <= multi_out;
		end if;
	end process;


	state_logic: process (RESET, CLK, EN) is
	begin
		if (RESET='1') then
			state <= state_start;
		elsif (CLK'event) and (CLK='1') then
			state <= next_state;
		end if;
	end process;

	fsm_logic: process (state, EN, OUT_BUSY, IN_VLD, CODE_DATA) is
	begin

	prog_count_inc <= '0';
	prog_count_dec <= '0';
	pointer_inc <= '0';
	pointer_dec <= '0';
	multi_sel <= "00";


	case state is
		when state_start =>
			next_state <= state_fetch;
		when state_fetch =>
			CODE_EN <= '1';
			DATA_EN <= '0';
			next_state <= state_decode;
		when state_decode =>
			case CODE_DATA is
				when "00111110" =>
					next_state <= state_pointer_inc;
				when "00111100" =>
					next_state <= state_pointer_dec;
				when "00101011" =>
					next_state <= state_value_inc;
				when "00101101" =>
					next_state <= state_value_dec;
				when "00101110" =>
					next_state <= state_putchar;
				when "00101100" =>
					next_state <= state_getchar;
				when "01011011" =>
					next_state <= state_cycle_start;
				when "01011101" =>
					next_state <= state_cycle_end;
				when "01111110" =>
					next_state <= state_cycle_break;
				when "00000000" =>
					next_state <= state_halt;
				when others =>
					next_state <= state_decode;
					prog_count_inc <= '1';
			end case;

		when state_pointer_inc =>
			pointer_inc <= '1';
			next_state <= state_fetch;
			prog_count_inc <= '1';

		when state_pointer_dec =>
			pointer_dec <= '1';
			next_state <= state_fetch;
			prog_count_inc <= '1';
		
		when state_value_inc =>
			DATA_EN <= '1';
			DATA_WREN <= '0';
			next_state <= state_value_inc_2;
		when state_value_inc_2 =>
			multi_sel <= "10";
			next_state <= state_value_inc_3;
		when state_value_inc_3 =>
			DATA_WREN <= '1';
			next_state <= state_value_inc_4;
		when state_value_inc_4 =>
			next_state <= state_fetch;
			prog_count_inc <= '1';
			
		when state_value_dec =>
			DATA_EN <= '1';
			DATA_WREN <= '0';
			next_state <= state_value_dec_2;
		when state_value_dec_2 =>
			multi_sel <= "01";
			next_state <= state_value_dec_3;
		when state_value_dec_3 =>
			DATA_WREN <= '1';
			next_state <= state_value_dec_4;
		when state_value_dec_4 =>
			next_state <= state_fetch;
			prog_count_inc <= '1';

		when state_putchar =>
			DATA_EN <= '1';
			DATA_WREN <= '0';
			next_state <= state_putchar_2;
		when state_putchar_2 =>
			if OUT_BUSY = '1'then
				DATA_EN <= '1';
				DATA_WREN <= '0';
				next_state <= state_putchar_2;
			else
				OUT_WREN <= '1';
				OUT_DATA <= DATA_RDATA;
				prog_count_inc <= '1';
				next_state <= state_fetch;
			end if;

		when state_getchar =>
			IN_REQ <= '1';
			multi_sel <= "00";
			next_state <= state_getchar_2;
		when state_getchar_2 =>
			if IN_VLD = '0' then
				IN_REQ <= '1';
				multi_sel <= "00";
				next_state <= state_getchar_2;
			else
				DATA_EN <= '1';
				DATA_WREN <= '1';
				prog_count_inc <= '1';
				next_state <= state_fetch;
			end if;
		
		when state_halt =>
			DATA_EN <= '0';
			OUT_WREN <= '0';

		when state_cycle_start =>
			prog_count_inc <= '1';
			DATA_EN <= '1';
			DATA_WREN <= '0';
			next_state <= state_cycle_start_1;
		when state_cycle_start_1 =>
			if DATA_RDATA = "00000000" then
				next_state <= state_cycle_start_2;
			else
				next_state <= state_fetch;
			end if;
		when state_cycle_start_2 =>
			temp <= CODE_DATA;
			prog_count_inc <= '1';
			if temp = "01011101" then
				next_state <= state_fetch;
	      	        else
				next_state <= state_cycle_start_2;
			end if;

		when state_cycle_end =>
			DATA_EN <= '1';
			DATA_WREN <= '0';
			next_state <= state_cycle_end_1;
		when state_cycle_end_1 =>
			if DATA_RDATA /= "00000000" then
				next_state <= state_cycle_end_2;
			else
				next_state <= state_fetch;
				prog_count_inc <= '1';
			end if;
		when state_cycle_end_2 =>
			prog_count_dec <= '1';
			next_state <= state_cycle_end_3;
		when state_cycle_end_3 =>
			temp <= CODE_DATA;
			next_state <= state_cycle_end_4;
		when state_cycle_end_4 =>
			if temp = "01011011" then
				next_state <= state_fetch;
				prog_count_inc <= '1';
	      	        else
				next_state <= state_cycle_end_2;
			end if;

		when state_cycle_break =>
			prog_count_inc <= '1';
			next_state <= state_cycle_break_0;
		when state_cycle_break_0 =>
			next_state <= state_cycle_break_1;
		when state_cycle_break_1 =>
			temp <= CODE_DATA;
			if temp = "01011101" then
				next_state <= state_fetch;
	      	        else
				prog_count_inc <= '1';
				next_state <= state_cycle_break_0;
			end if;
		
	end case;
	end process;



 -- pri tvorbe kodu reflektujte rady ze cviceni INP, zejmena mejte na pameti, ze 
 --   - nelze z vice procesu ovladat stejny signal,
 --   - je vhodne mit jeden proces pro popis jedne hardwarove komponenty, protoze pak
 --      - u synchronnich komponent obsahuje sensitivity list pouze CLK a RESET a 
 --      - u kombinacnich komponent obsahuje sensitivity list vsechny ctene signaly.
end behavioral;
